\begin{frame}{Slide 2: Problem Statement and Motivation}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Challenges in NCRB Crime Registries}
                \begin{itemize}
                    \item \textbf{Extreme Spatial Skew:} Minority of urban hubs account for a disproportionately large volume of crimes.
                    \item \textbf{State Reporting Disparities:} Variation in police reporting cultures and chargesheeting standards.
                    \item \textbf{High-Dimensional Complexity:} Hundreds of overlapping heads across IPC, SLL, cyber, and vulnerable groups.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Motivation \& Policy Need}
                \begin{itemize}
                    \item \textbf{Heterogeneous Scale:} Raw crime totals distort risk profiling across 964 diverse districts.
                    \item \textbf{Evidence-Based Allocation:} Need to benchmark metropolitan policing and isolate localized outliers.
                    \item \textbf{Actionable AI:} Provide law enforcement with conversational, data-grounded intelligence.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.3cm}
    \begin{alertblock}{Analytical Scope}
        Evaluating \textbf{964 Districts} across \textbf{36 States/UTs} and \textbf{37 Million-Plus Cities} (NCRB 2024 Release).
    \end{alertblock}
\end{frame}

% ==============================================================================
% SLIDE 3: OBJECTIVES
% ==============================================================================
\begin{frame}{Slide 3: Objectives}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Core Technical Objectives}
                \begin{itemize}
                    \item \textbf{Dual-Tier Ingestion:} Ingest \& clean 17 audited NCRB datasets into master feature matrices.
                    \item \textbf{Feature Engineering:} Derive 25 per-capita composite vulnerability indices normalized via RobustScaler.
                    \item \textbf{Unsupervised ML Pipeline:} Execute PCA dimensionality reduction, K-Means clustering ($k=5$), and DBSCAN anomaly detection.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Applied Policy \& Deployment Objectives}
                \begin{itemize}
                    \item \textbf{Metro Efficiency Benchmark:} Construct a 4-quadrant efficiency matrix across 37 million-plus cities.
                    \item \textbf{Causal Motive Analysis:} Decompose 35 murder and 25 cybercrime motive heads.
                    \item \textbf{Production Dashboard:} Deploy a glassmorphic Streamlit web suite with Groq LLaMA 3.3 AI.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \begin{block}{Target Deliverable}
        An end-to-end, reproducible Big Data Analytics platform transforming unstructured government registries into actionable spatial insights.
    \end{block}
\end{frame}

% ==============================================================================
% SLIDE 4: EXISTING SYSTEM / LITERATURE REVIEW
% ==============================================================================
\begin{frame}{Slide 4: Existing System / Literature Review}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Existing NCRB Reporting Limitations}
                \begin{itemize}
                    \item \textbf{Static Aggregations:} Published primarily as static PDF/Excel tables without multi-dimensional cross-filtering.
                    \item \textbf{Raw Volume Distortion:} Highly populated districts dominate absolute volume counts without per-capita normalization.
                    \item \textbf{Lack of Unsupervised ML:} Absence of automated spatial clustering and density-based anomaly discovery.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Literature \& Methodological Gap}
                \begin{itemize}
                    \item \textbf{Decoupled Hierarchy:} Existing studies mix rural districts and megacities without a dual-tier framework.
                    \item \textbf{Outlier Sensitivity:} Standard MinMax scaling suffers from extreme district skew; addressed here by \textbf{RobustScaler}.
                    \item \textbf{Integrated AI:} Prior platforms lack guardrailed LLM conversational assistants for non-technical planners.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
